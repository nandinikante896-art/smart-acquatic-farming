def find_best_model(root: Path, preferred_names=None):

    if not root.exists():
        return None

    preferred_names = preferred_names or []

    # First check preferred locations
    for name in preferred_names:
        p = root / name
        if p.exists():
            return p

    # Search recursively
    candidates = list(root.rglob("best.pt"))

    if not candidates:
        candidates = list(root.rglob("*.pt"))

    if not candidates:
        return None

    # Prefer trained/final models
    priority_words = [
        "final",
        "disease",
        "trained",
        "best"
    ]

    candidates = sorted(
        candidates,
        key=lambda p: (
            sum(word in str(p).lower() for word in priority_words),
            p.stat().st_mtime
        ),
        reverse=True
    )

    return candidates[0]

def image_to_base64(image):

    buffer = io.BytesIO()

    image.save(buffer, format="JPEG", quality=92)

    return (
        "data:image/jpeg;base64,"
        + base64.b64encode(buffer.getvalue()).decode("utf-8")
    )


def read_image(file):

    image = Image.open(file)

    # Always RGB
    if image.mode != "RGB":
        image = image.convert("RGB")

    # Avoid extremely huge images
    max_side = 1600

    if max(image.size) > max_side:
        image.thumbnail((max_side, max_side))

    return image


def get_font(size=22):

    possible_fonts = [
        r"C:\Windows\Fonts\arial.ttf",
        r"C:\Windows\Fonts\segoeui.ttf",
    ]

    for font in possible_fonts:
        if os.path.exists(font):
            try:
                return ImageFont.truetype(font, size)
            except Exception:
                pass

    return ImageFont.load_default()

def analyse_do(value):

    try:
        value = float(value)
    except Exception:
        return {
            "value": None,
            "level": "Invalid",
            "message": "Enter a valid DO value."
        }

    if value >= 5:
        level = "Optimal"
        message = "Dissolved oxygen is in the project's optimal screening range."
    elif value >= 3:
        level = "Moderate"
        message = "Monitor oxygen and pond activity."
    elif value >= 2:
        level = "Low"
        message = "Consider increasing aeration and monitoring closely."
    else:
        level = "Critical"
        message = "Immediate attention and aeration monitoring are recommended."

    return {
        "value": value,
        "level": level,
        "message": message
    }

# YOLO DETECTION
# ============================================================

def detect_image(image, animal):

    animal = animal.lower()

    if animal not in ["fish", "prawn"]:
        animal = "fish"

    model = loaded_models.get(animal)

    if model is None:
        return {
            "success": False,
            "error": f"{animal.title()} disease model was not found.",
            "detections": [],
            "image": image_to_base64(image)
        }

    classes = (
        FISH_CLASSES
        if animal == "fish"
        else PRAWN_CLASSES
    )

    try:

        results = model.predict(
            source=image,
            conf=0.20,
            imgsz=640,
            device="cpu",
            verbose=False
        )

        result = results[0]

        output = image.copy()
        draw = ImageDraw.Draw(output)

        font = get_font(20)

        detections = []

        if result.boxes is not None and len(result.boxes) > 0:

            for box in result.boxes:

                xyxy = box.xyxy[0].cpu().tolist()

                x1, y1, x2, y2 = [
                    int(max(0, v))
                    for v in xyxy
                ]

                confidence = float(
                    box.conf[0].cpu().item()
                )

                class_id = int(
                    box.cls[0].cpu().item()
                )

                label = classes.get(
                    class_id,
                    f"Class {class_id}"
                )

                # Only actual detected object gets box
                draw.rectangle(
                    [x1, y1, x2, y2],
                    outline=(220, 30, 30),
                    width=5
                )

                text_label = f"{label}  {confidence * 100:.1f}%"

                bbox = draw.textbbox(
                    (x1, y1),
                    text_label,
                    font=font
                )

                tw = bbox[2] - bbox[0]
                th = bbox[3] - bbox[1]

                text_y = max(0, y1 - th - 10)

                draw.rounded_rectangle(
                    [
                        x1,
                        text_y,
                        x1 + tw + 16,
                        text_y + th + 10
                    ],
                    radius=6,
                    fill=(220, 30, 30)
                )

                draw.text(
                    (x1 + 8, text_y + 5),
                    text_label,
                    fill="white",
                    font=font
                )

                detections.append({
                    "class_id": class_id,
                    "label": label,
                    "confidence": round(confidence, 4),
                    "confidence_percent": round(
                        confidence * 100, 1
                    ),
                    "box": [
                        x1, y1, x2, y2
                    ]
                })

        return {
            "success": True,
            "animal": animal,
            "detections": detections,
            "count": len(detections),
            "image": image_to_base64(output),
            "message": (
                "Detection completed."
                if detections
                else "No detection"
            )
        }

    except Exception as e:

        traceback.print_exc()

        return {
            "success": False,
            "error": str(e),
            "detections": [],
            "image": image_to_base64(image)
        }

@app.route("/api/detect", methods=["POST"])
def api_detect():

    if "image" not in request.files:

        return jsonify({
            "success": False,
            "error": "No image received."
        }), 400

    file = request.files["image"]

    animal = request.form.get(
        "animal",
        "fish"
    )

    if not file.filename:

        return jsonify({
            "success": False,
            "error": "No image selected."
        }), 400

    try:

        image = read_image(file)

        result = detect_image(
            image,
            animal
        )

        return jsonify(result)

    except Exception as e:

        traceback.print_exc()

        return jsonify({
            "success": False,
            "error": str(e)
        }), 500

@app.route("/api/farmer", methods=["POST"])
def api_farmer():

    data = request.get_json(
        silent=True
    ) or {}

    question = str(
        data.get("question", "")
    ).strip()

    if not question:

        return jsonify({
            "success": False,
            "answer": "Please enter your question."
        })

    return jsonify({
        "success": True,
        "answer": farmer_answer(question)
    })

async function runDetection() {

    const input =
        document.getElementById(
            "imageInput"
        );

    if (!input.files.length) {

        alert(
            "Please upload an image first."
        );

        return;
    }

    const file =
        input.files[0];

    const form =
        new FormData();

    form.append(
        "image",
        file
    );

    form.append(
        "animal",
        selectedAnimal
    );


    document.getElementById(
        "resultText"
    ).textContent =
        "Running detection...";


    try {

        const response =
            await fetch(
                "/api/detect",
                {
                    method: "POST",
                    body: form
                }
            );

        const data =
            await response.json();


        if (!data.success) {

            document.getElementById(
                "resultText"
            ).textContent =
                data.error ||
                "Detection failed.";

            return;
        }


        document.getElementById(
            "previewContainer"
        ).innerHTML = `
            <img
                src="${data.image}"
                class="preview"
                alt="Detection result">
        `;


        if (data.count === 0) {

            document.getElementById(
                "resultText"
            ).textContent =
                "No detection";

            document.getElementById(
                "detectionResults"
            ).innerHTML = `
                <div class="no-detection">
                    ⚠️ No object was detected by the
                    selected model.
                    <br><br>
                    Try a clearer image similar to the
                    training data.
                </div>
            `;

            return;
        }


        document.getElementById(
            "resultText"
        ).textContent =
            `${data.count} detection(s) found`;


        let html = "";

        data.detections.forEach(
            item => {

                html += `
                    <div class="detection-item">

                        <div>
                            <strong>
                                ${item.label}
                            </strong>

                            <div class="muted">
                                Bounding box:
                                ${item.box.join(", ")}
                            </div>
                        </div>

                        <div class="confidence">
                            ${item.confidence_percent}%
                        </div>

                    </div>
                `;

            }
        );


        document.getElementById(
            "detectionResults"
        ).innerHTML = html;


    } catch (error) {

        console.error(error);

        document.getElementById(
            "resultText"
        ).textContent =
            "Server connection error.";

    }

}

async function startCamera(mode) {

    stopCamera();

    try {

        cameraStream =
            await navigator.mediaDevices.getUserMedia({
                video: {
                    facingMode: {
                        ideal: mode
                    },
                    width: {
                        ideal: 1280
                    },
                    height: {
                        ideal: 720
                    }
                },
                audio: false
            });


        const video =
            document.getElementById(
                "cameraVideo"
            );

        video.srcObject =
            cameraStream;

        document.getElementById(
            "cameraPlaceholder"
        ).style.display = "none";


    } catch (error) {

        console.error(error);

        alert(
            "Camera permission denied or camera is unavailable."
        );

    }

}


function stopCamera() {

    if (cameraStream) {

        cameraStream
            .getTracks()
            .forEach(track => {
                track.stop();
            });

        cameraStream = null;

    }

    const video =
        document.getElementById(
            "cameraVideo"
        );

    video.srcObject = null;

    document.getElementById(
        "cameraPlaceholder"
    ).style.display = "block";

}


function captureCamera() {

    const video =
        document.getElementById(
            "cameraVideo"
        );

    if (!cameraStream) {

        alert(
            "Start the camera first."
        );

        return;
    }


    const canvas =
        document.getElementById(
            "cameraCanvas"
        );

    canvas.width =
        video.videoWidth;

    canvas.height =
        video.videoHeight;


    const ctx =
        canvas.getContext("2d");

    ctx.drawImage(
        video,
        0,
        0,
        canvas.width,
        canvas.height
    );


    canvas.toBlob(
        async blob => {

            const form =
                new FormData();

            form.append(
                "image",
                blob,
                "camera_capture.jpg"
            );

            form.append(
                "animal",
                selectedAnimal
            );


            document.getElementById(
                "cameraResult"
            ).innerHTML = `
                <div class="muted">
                    Detecting captured image...
                </div>
            `;


            try {

                const response =
                    await fetch(
                        "/api/detect",
                        {
                            method: "POST",
                            body: form
                        }
                    );

                const data =
                    await response.json();


                if (!data.success) {

                    document.getElementById(
                        "cameraResult"
                    ).innerHTML = `
                        <div class="no-detection">
                            ${data.error}
                        </div>
                    `;

                    return;
                }


                let resultHtml = `
                    <img
                        src="${data.image}"
                        class="preview">
                `;


                if (data.count === 0) {

                    resultHtml += `
                        <div class="no-detection"
                             style="margin-top:12px">

                            ⚠️ No detection

                        </div>
                    `;

                } else {

                    data.detections.forEach(
                        item => {

                            resultHtml += `
                                <div class="detection-item"
                                     style="margin-top:8px">

                                    <strong>
                                        ${item.label}
                                    </strong>

                                    <span class="confidence">
                                        ${item.confidence_percent}%
                                    </span>

                                </div>
                            `;

                        }
                    );

                }


                document.getElementById(
                    "cameraResult"
                ).innerHTML =
                    resultHtml;


            } catch (error) {

                console.error(error);

                document.getElementById(
                    "cameraResult"
                ).innerHTML = `
                    <div class="no-detection">
                        Camera detection failed.
                    </div>
                `;

            }

        },
        "image/jpeg",
        0.92
    );

}

            await fetch(

                            "application/json"

                    body: JSON.stringify({
                        question: question
                    })
                }
            );

            await response.json();


        lastAssistantAnswer =
            data.answer || "";


        addMessage(
            lastAssistantAnswer,
            "bot"
        );


    } catch (error) {

        addMessage(
            "Unable to connect to AquaCare assistant.",
            "bot"
        );

    }

}

function startVoice() {

    const SpeechRecognition =
        window.SpeechRecognition ||
        window.webkitSpeechRecognition;


    if (!SpeechRecognition) {

        alert(
            "Voice input is not supported by this browser."
        );

        return;
    }


    const language =
        document.getElementById(
            "language"
        ).value;


    const recognition =
        new SpeechRecognition();


    const langMap = {

        en: "en-IN",
        te: "te-IN",
        hi: "hi-IN",
        ta: "ta-IN",
        kn: "kn-IN"

    };


    recognition.lang =
        langMap[language] ||
        "en-IN";


    recognition.interimResults =
        false;


    recognition.onresult =
        event => {

            const transcript =
                event.results[0][0].transcript;

            document.getElementById(
                "chatInput"
            ).value =
                transcript;

        };


    recognition.onerror =
        error => {

            console.log(
                "Speech error:",
                error
            );

        };


    recognition.start();

}


function speakLastAnswer() {

    if (!lastAssistantAnswer) {

        alert(

        );

        return;
    }


    if (!window.speechSynthesis) {

        alert(

        ).value;


    const langMap = {

        en: "en-IN",
        te: "te-IN",
        hi: "hi-IN",
        ta: "ta-IN",
        kn: "kn-IN"

    };


    const speech =
        new SpeechSynthesisUtterance(
            lastAssistantAnswer
        );


    speech.lang =
        langMap[language] ||
        "en-IN";


    window.speechSynthesis.cancel();

    window.speechSynthesis.speak(
        speech
    );

}
